cmake_minimum_required(VERSION 3.16)
project(slay_the_nspire_2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(spire_core STATIC src/core.cpp src/relics.cpp src/run.cpp src/content.cpp src/app.cpp src/pointer.cpp src/reference_rng.cpp src/reference_map.cpp src/map_preview.cpp src/reference_math.cpp src/reference_act.cpp src/map_preview_input.cpp)
target_include_directories(spire_core PUBLIC include)
option(SPIRE_X11 "Build the Linux X11 desktop host" ON)
if(SPIRE_X11)
  find_package(X11 REQUIRED)
  add_executable(nSpire2 src/host_x11.cpp)
  target_link_libraries(nSpire2 PRIVATE spire_core X11::X11)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(core_regression tests/core_regression.cpp)
  target_link_libraries(core_regression PRIVATE spire_core)
  add_test(NAME core_regression COMMAND core_regression "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/v3-suspended.hex")
  add_executable(input_regression tests/input_regression.cpp)
  target_link_libraries(input_regression PRIVATE spire_core)
  add_test(NAME input_regression COMMAND input_regression)
endif()
if(BUILD_TESTING)
  add_executable(content_regression tests/content_regression.cpp)
  target_link_libraries(content_regression PRIVATE spire_core)
  add_test(NAME content_regression COMMAND content_regression)
endif()
