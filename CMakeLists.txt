cmake_minimum_required(VERSION 3.16)
project(slay_the_nspire_2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(spire_core STATIC src/core.cpp src/relics.cpp src/run.cpp src/content.cpp src/app.cpp src/pointer.cpp src/reference_rng.cpp src/reference_map.cpp src/map_preview.cpp)
target_include_directories(spire_core PUBLIC include)
option(SPIRE_X11 "Build the Linux X11 desktop host" ON)
if(SPIRE_X11)
  find_package(X11 REQUIRED)
  add_executable(nSpire2 src/host_x11.cpp)
  target_link_libraries(nSpire2 PRIVATE spire_core X11::X11)
endif()
